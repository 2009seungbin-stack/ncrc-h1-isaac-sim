    """Update configuration for RSL-RL agent based on inputs.
    Args:
        agent_cfg: The configuration for RSL-RL agent.

    """
    # override the default configuration with CLI arguments
    if hasattr(args_cli, "seed") and args_cli.seed is not None:

    if args_cli.load_run is not None:

    if args_cli.checkpoint is not None:

def _model_iter(p: Path) -> int:
    # model_<iter>.pt → iter. 비숫자(model_best.pt 등 resume 용 사용자 파일)는

def _is_run(p: Path) -> bool:

def find_latest_run() -> Path:
    if not LOG_ROOT.exists():
        raise FileNotFoundError(f"학습 log 없음: {LOG_ROOT}")
    runs = [p for p in LOG_ROOT.iterdir() if p.is_dir()]
    if not runs:
        raise FileNotFoundError(f"학습 run 없음: {LOG_ROOT}")
    real = [p for p in runs if _is_run(p)]
    if not real:
        # ⚠️ '학습 기록이 없다' 와 '체크포인트가 없다' 는 다른 말이다 — 섞으면
        #    참가자가 체크포인트를 찾아 헤맨다. 무엇이 없는지 그대로 적는다.
        raise FileNotFoundError(
            f"학습 기록(tfevents)이 있는 run 이 없습니다: {LOG_ROOT}"
            f" — 학습이 시작되지 못했습니다. 이어서 학습용으로 만든 폴더"
            f"(예: from_best)는 run 이 아닙니다. 위쪽 학습 로그의 첫 오류를 보세요.")
    return max(real, key=lambda p: p.stat().st_mtime)
